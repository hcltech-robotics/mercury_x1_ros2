cmake_minimum_required(VERSION 3.8)
project(mercury_x1_bringup)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)

# arm-aware Nav2 footprint (started by launch/nav.launch.py)
add_executable(arm_footprint src/arm_footprint.cpp)
ament_target_dependencies(arm_footprint rclcpp tf2 tf2_ros geometry_msgs ament_index_cpp)
target_link_libraries(arm_footprint yaml-cpp)

install(TARGETS arm_footprint DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
